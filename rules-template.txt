rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function google() {return request.auth != null && request.auth.token.email_verified == true && request.auth.token.firebase.sign_in_provider == 'google.com';}
    function owner() {return google() && request.auth.token.email == 'CHANGE_ME@example.com';}
    function allowed(email) {return email == 'CHANGE_ME@example.com' || (exists(/databases/$(database)/documents/sparkAccess/$(email)) && get(/databases/$(database)/documents/sparkAccess/$(email)).data.enabled == true);}
    function teacher(uid) {return google() && request.auth.uid == uid && allowed(request.auth.token.email);}
    function guest() {return request.auth != null && request.auth.token.firebase.sign_in_provider == 'anonymous';}
    function room(pin) {return get(/databases/$(database)/documents/liveRooms/$(pin)).data;}
    function ownRoom(r) {return teacher(r.teacherUid) && request.auth.token.email == r.teacherEmail;}
    function member(pin) {return guest() && exists(/databases/$(database)/documents/liveRooms/$(pin)/players/$(request.auth.uid));}
    function soloAvailable(bankId) {return !exists(/databases/$(database)/documents/soloLocks/$(bankId)) || get(/databases/$(database)/documents/soloLocks/$(bankId)).data.activeRooms.size() == 0;}
    match /soloLocks/{bankId} {
      allow get: if teacher(get(/databases/$(database)/documents/soloBanks/$(bankId)).data.teacherUid) || (guest() && allowed(get(/databases/$(database)/documents/soloBanks/$(bankId)).data.teacherEmail));
      allow list: if false;
      allow create: if teacher(request.resource.data.teacherUid) && request.resource.data.teacherEmail == request.auth.token.email
        && request.resource.data.keys().hasOnly(['teacherUid','teacherEmail','activeRooms']) && request.resource.data.activeRooms is list;
      allow update: if teacher(resource.data.teacherUid) && request.resource.data.teacherUid == resource.data.teacherUid
        && request.resource.data.teacherEmail == resource.data.teacherEmail && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['activeRooms']) && request.resource.data.activeRooms is list;
      allow delete: if false;
    }
    match /sparkAccess/{email} {
      allow get: if owner() || (google() && request.auth.token.email == email);
      allow list: if owner();
      allow create, update: if owner() && email != 'CHANGE_ME@example.com'
        && request.resource.data.keys().hasOnly(['email','enabled','label','updatedAt'])
        && request.resource.data.email == email && request.resource.data.enabled is bool
        && request.resource.data.label is string && request.resource.data.label.size() <= 100
        && request.resource.data.updatedAt == request.time;
      allow delete: if false;
    }
    match /sparkTeachers/{uid}/{document=**} {
      allow read, write: if teacher(uid);
    }
    match /soloLinks/{uid} {
      allow get: if teacher(uid) || (guest() && allowed(resource.data.teacherEmail));
      allow list: if false;
      allow create,update: if teacher(uid) && request.resource.data.teacherUid == uid
        && request.resource.data.teacherEmail == request.auth.token.email
        && request.resource.data.keys().hasOnly(['teacherUid','teacherEmail','archiveId']);
      allow delete: if teacher(uid);
    }
    match /soloBanks/{bankId} {
      function bankOwner() {return teacher(resource.data.teacherUid) && request.auth.token.email == resource.data.teacherEmail;}
      function bankTeacher() {let b=getAfter(/databases/$(database)/documents/soloBanks/$(bankId)).data;return teacher(b.teacherUid) && request.auth.token.email == b.teacherEmail;}
      allow get: if bankOwner() || (guest() && allowed(resource.data.teacherEmail)
        && (!raceBank(bankId) || (member(bankId) && room(bankId).phase == 'question')));
      allow list: if false;
      allow create: if teacher(request.resource.data.teacherUid) && request.resource.data.teacherEmail == request.auth.token.email
        && request.resource.data.keys().hasOnly(['teacherUid','teacherEmail','questions']);
      allow update: if bankOwner() && request.resource.data == resource.data;
      allow delete: if false;
      match /keys/{index} {
        allow create: if bankTeacher() && request.resource.data.keys().hasOnly(['correct']) && request.resource.data.correct is int && request.resource.data.correct >= 0 && request.resource.data.correct <= 3;
        allow update: if bankTeacher() && request.resource.data == resource.data;
        allow get: if bankTeacher() || (guest() && soloAvailable(bankId)
          && exists(/databases/$(database)/documents/soloBanks/$(bankId)/unlocks/$(request.auth.uid + '_' + index))
          && allowed(get(/databases/$(database)/documents/soloBanks/$(bankId)).data.teacherEmail));
        allow list,delete: if false;
      }
      match /unlocks/{submissionId} {
        allow create: if guest() && soloAvailable(bankId) && submissionId == request.auth.uid + '_' + string(request.resource.data.index)
          && request.resource.data.keys().hasOnly(['studentUid','index','attemptId'])
          && request.resource.data.studentUid == request.auth.uid && request.resource.data.index is int
          && request.resource.data.index >= 0 && request.resource.data.index < 15
          && get(/databases/$(database)/documents/soloAttempts/$(request.resource.data.attemptId)).data.studentUid == request.auth.uid
          && get(/databases/$(database)/documents/soloAttempts/$(request.resource.data.attemptId)).data.bankId == bankId
          && exists(/databases/$(database)/documents/soloAttempts/$(request.resource.data.attemptId)/answers/$(string(request.resource.data.index)))
          && allowed(get(/databases/$(database)/documents/soloBanks/$(bankId)).data.teacherEmail);
        allow get: if bankTeacher() || (guest() && resource.data.studentUid == request.auth.uid && allowed(get(/databases/$(database)/documents/soloBanks/$(bankId)).data.teacherEmail));
        allow list: if bankTeacher();
        allow update,delete: if false;
      }
    }
    function raceBank(id) {
      return exists(/databases/$(database)/documents/liveRooms/$(id))
        && room(id).get('mode','sync') == 'race';
    }
    function attemptPlayable(a, id) {
      return soloAvailable(a.bankId) && (id == a.bankId + '_' + a.studentUid
        ? raceBank(a.bankId) && room(a.bankId).phase == 'question' && member(a.bankId)
        : !raceBank(a.bankId));
    }
    match /soloAttempts/{attemptId} {
      function ownAttempt(a) {return guest() && request.auth.uid == a.studentUid && allowed(a.teacherEmail);}
      function attemptTeacher(a) {return teacher(a.teacherUid) && request.auth.token.email == a.teacherEmail;}
      function stable(a,b) {return b.studentUid == a.studentUid && b.teacherUid == a.teacherUid && b.teacherEmail == a.teacherEmail && b.bankId == a.bankId && b.name == a.name && b.gender == a.gender && b.created == a.created;}
      function grade(a,b) {
        let choice=get(/databases/$(database)/documents/soloAttempts/$(attemptId)/answers/$(string(a.index))).data;
        let key=get(/databases/$(database)/documents/soloBanks/$(a.bankId)/keys/$(string(a.index))).data;
        let elapsed=choice.submittedAt - a.currentStartedAt;
        let right=choice.answer == key.correct && elapsed >= duration.value(0,'s') && elapsed < duration.value(30,'s');
        let rawBonus=500-elapsed.seconds()*25;
        let bonus=rawBonus < 0 ? 0 : (rawBonus > 500 ? 500 : rawBonus);
        return b.diff(a).affectedKeys().hasOnly(['level','score','speedPoints','status'])
          && b.status == (a.index == 14 ? 'finished' : 'answered')
          && b.level == a.level + (right ? 1 : 0)
          && b.score == a.score + (right ? 1000+bonus : 0)
          && b.speedPoints == a.speedPoints + (right ? bonus : 0);
      }
      function initialAttempt() {return request.resource.data.keys().hasOnly(['studentUid','teacherUid','teacherEmail','bankId','name','gender','created','index','currentStartedAt','level','score','speedPoints','status'])
        && get(/databases/$(database)/documents/soloBanks/$(request.resource.data.bankId)).data.teacherUid == request.resource.data.teacherUid
        && get(/databases/$(database)/documents/soloBanks/$(request.resource.data.bankId)).data.teacherEmail == request.resource.data.teacherEmail
        && request.resource.data.name is string && request.resource.data.name.size() >= 2 && request.resource.data.name.size() <= 80
        && request.resource.data.gender in ['male','female'] && request.resource.data.created == request.time
        && request.resource.data.index == 0 && request.resource.data.currentStartedAt == null
        && request.resource.data.level == 0 && request.resource.data.score == 0 && request.resource.data.speedPoints == 0 && request.resource.data.status == 'intro';}
      allow create: if initialAttempt() && (
        (ownAttempt(request.resource.data) && soloAvailable(request.resource.data.bankId) && !raceBank(request.resource.data.bankId))
        || (attemptTeacher(request.resource.data) && raceBank(request.resource.data.bankId)
          && room(request.resource.data.bankId).phase == 'preparing'
          && attemptId == request.resource.data.bankId + '_' + request.resource.data.studentUid)
      );
      allow get,list: if ownAttempt(resource.data) || attemptTeacher(resource.data);
      allow update: if ownAttempt(resource.data) && attemptPlayable(resource.data, attemptId) && stable(resource.data,request.resource.data)
        && ((resource.data.status == 'intro' && request.resource.data.status == 'playing'
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','currentStartedAt']) && request.resource.data.currentStartedAt == request.time)
          || (resource.data.status == 'playing' && grade(resource.data,request.resource.data))
          || (resource.data.status == 'answered' && request.resource.data.status == 'playing'
              && request.resource.data.index == resource.data.index+1 && request.resource.data.index < 15
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','index','currentStartedAt']) && request.resource.data.currentStartedAt == request.time));
      allow delete: if false;
      match /answers/{index} {
        function attempt() {return get(/databases/$(database)/documents/soloAttempts/$(attemptId)).data;}
        allow create: if ownAttempt(attempt()) && attemptPlayable(attempt(), attemptId) && attempt().status == 'playing' && index == string(attempt().index)
          && request.resource.data.keys().hasOnly(['answer','submittedAt']) && request.resource.data.answer is int && request.resource.data.answer >= -1 && request.resource.data.answer <= 3
          && request.resource.data.submittedAt == request.time;
        allow get,list: if ownAttempt(attempt()) || attemptTeacher(attempt());
        allow update,delete: if false;
      }
    }
    match /liveRooms/{pin} {
      function validPublic(r) {
        return r.keys().hasOnly(['teacherUid','teacherEmail','phase','index','question','correct','players','phaseStartedAt','created','mode'])
          && r.get('mode','sync') in ['sync','race']
          && r.phase in ['lobby','preparing','question','revealed','finished'] && r.index is int && r.index >= 0 && r.index <= 15
          && r.players is list && r.players.size() <= 100
          && ((r.phase == 'revealed' && r.correct is int && r.correct >= 0 && r.correct <= 3) || (r.phase != 'revealed' && r.correct == null))
          && (r.question == null || (r.question is map && r.question.keys().hasOnly(['text','options']) && r.question.text is string && r.question.options is list && r.question.options.size() == 4));
      }
      allow create: if ownRoom(request.resource.data) && validPublic(request.resource.data)
        && request.resource.data.phase == 'lobby' && request.resource.data.index == 0
        && request.resource.data.created == request.time;
      allow update: if ownRoom(resource.data) && validPublic(request.resource.data)
        && request.resource.data.teacherUid == resource.data.teacherUid
        && request.resource.data.teacherEmail == resource.data.teacherEmail
        && request.resource.data.created == resource.data.created
        && request.resource.data.get('mode','sync') == resource.data.get('mode','sync')
        && (request.resource.data.phase != 'question' || request.resource.data.phaseStartedAt == request.time);
      allow delete: if ownRoom(resource.data);
      allow get: if ownRoom(resource.data) || (member(pin) && allowed(resource.data.teacherEmail));
      allow list: if ownRoom(resource.data);
      match /players/{studentUid} {
        allow create: if guest() && request.auth.uid == studentUid && room(pin).phase == 'lobby' && allowed(room(pin).teacherEmail)
          && request.resource.data.keys().hasOnly(['id','name','gender','avatar','joinedAt'])
          && request.resource.data.id == studentUid && request.resource.data.name is string
          && request.resource.data.name.size() >= 2 && request.resource.data.name.size() <= 80
          && request.resource.data.gender in ['male','female'] && request.resource.data.avatar is int
          && request.resource.data.avatar >= 0 && request.resource.data.avatar <= 7
          && request.resource.data.joinedAt == request.time;
        allow get: if ownRoom(room(pin)) || (guest() && request.auth.uid == studentUid && allowed(room(pin).teacherEmail));
        allow list: if ownRoom(room(pin));
        allow update, delete: if false;
      }
      match /answers/{answerId} {
        allow create: if guest() && member(pin) && allowed(room(pin).teacherEmail)
          && request.resource.data.keys().hasOnly(['studentUid','index','answer','submittedAt'])
          && request.resource.data.studentUid == request.auth.uid
          && request.resource.data.index == room(pin).index
          && answerId == request.auth.uid + '_' + string(room(pin).index)
          && room(pin).phase == 'question' && room(pin).get('mode','sync') == 'sync'
          && request.time < room(pin).phaseStartedAt + duration.value(30,'s')
          && request.resource.data.answer is int && request.resource.data.answer >= 0 && request.resource.data.answer <= 3
          && request.resource.data.submittedAt == request.time;
        allow get: if ownRoom(room(pin)) || (member(pin) && answerId == request.auth.uid + '_' + string(room(pin).index) && allowed(room(pin).teacherEmail));
        allow list: if ownRoom(room(pin));
        allow update, delete: if false;
      }
    }
    match /{document=**} {allow read,write: if false;}
  }
}
